#include <iostream>

// Функция для вывода динамического массива с учетом логического и фактического размеров
void print_dynamic_array(int* arr, int logical_size, int actual_size) {
    for (int i = 0; i < actual_size; ++i) {
        if (i < logical_size) {
            std::cout << arr[i] << " "; // Выводим значение элемента
        } else {
            std::cout << "_ "; // Если элемент "в запасе" — выводим '_'
        }
    }
    std::cout << std::endl;
}

// Функция добавления элемента в динамический массив с возможным увеличением размера массива
// arr - ссылка на указатель на массив, logical_size - ссылка на логический размер,
// actual_size - ссылка на фактический размер, value - добавляемое значение
void append_to_dynamic_array(int*& arr, int& logical_size, int& actual_size, int value) {
    // Если есть свободное место (логический размер меньше фактического)
    if (logical_size < actual_size) {
        arr[logical_size] = value;
        ++logical_size;
    } 
    // Если свободного места нет, увеличиваем массив в 2 раза
    else {
        int new_actual_size = actual_size * 2;
        // Если массив нулевого размера, делаем его размером 1
        if (new_actual_size == 0) new_actual_size = 1;
        int* new_arr = new int[new_actual_size];

        // Копируем старые элементы
        for (int i = 0; i < logical_size; ++i) {
            new_arr[i] = arr[i];
        }
        // Добавляем новый элемент в конец
        new_arr[logical_size] = value;
        ++logical_size;
        // Старый массив больше не нужен, освобождаем память
        delete[] arr;
        arr = new_arr;
        actual_size = new_actual_size;
    }
}

int main() {
    int actual_size;
    int logical_size;

    // Ввод фактического и логического размеров
    std::cout << "Введите фактичеcкий размер массива: ";
    std::cin >> actual_size;

    std::cout << "Введите логический размер массива: ";
    std::cin >> logical_size;

    // Проверка корректности размеров
    if (logical_size > actual_size || actual_size < 0 || logical_size < 0) {
        std::cout << "Ошибка: некорректные размеры массива." << std::endl;
        return 1;
    }

    // Выделяем память под массив
    int* arr = new int[actual_size];

    // Вводим значения массива
    for (int i = 0; i < logical_size; ++i) {
        std::cout << "Введите arr[" << i << "]: ";
        std::cin >> arr[i];
    }

    std::cout << "Динамический массив: ";
    print_dynamic_array(arr, logical_size, actual_size);

    // Постоянно принимаем новые значения, пока пользователь не введет 0
    while (true) {
        std::cout << "Введите элемент для добавления: ";
        int value;
        std::cin >> value;
        if (value == 0) break; // Пользователь завершил ввод

        append_to_dynamic_array(arr, logical_size, actual_size, value);
        std::cout << "Динамический массив: ";
        print_dynamic_array(arr, logical_size, actual_size);
    }

    std::cout << "Спасибо! Ваш массив: ";
    print_dynamic_array(arr, logical_size, actual_size);

    // Освобождаем память
    delete[] arr;

    return 0;
}

Ссылка (как в параметрах типа int& или int*& в функции append_to_dynamic_array) используется для того, чтобы функция могла изменять переменную, переданную ей из main, а не только её локальную копию. Вот почему это важно:

• Когда вы передаете переменную в функцию по значению (например, просто int), она копируется. Любые изменения этой копии никак не повлияют на величину, находящуюся в вызывающей программе.
• При работе с динамическим массивом и его размером нам нужно изменять и сам массив (возможно, выделяя новый и делая переназначение указателя), и размеры (logical_size и actual_size). Именно для того, чтобы изменения были видны в main, параметры передаются по ссылке.
• Особенно важно это для указателя на массив: если мы расширяем массив (добавляем элементы, увеличиваем actual_size), мы выделяем новый массив и делаем arr указывать на него. Поэтому мы используем int*& arr — "ссылка на указатель". Это позволяет перенаправить внешний указатель arr на новый массив прямо в функции.
• Для размеров (logical_size и actual_size) мы используем int& — ссылка на переменную int, чтобы их изменение в функции сохранялось и после возвращения из функции.

Когда мы пишем функцию с параметром `int*& arr`, мы передаём в неё ссылку на указатель — то есть не копию указателя, а сам оригинальный указатель из main. Это позволяет нам в функции делать две вещи:

1. Освободить память старого массива с помощью `delete[] arr`. Поскольку arr указывает на тот же самый участок памяти, что и в main, освобождается ровно та память, что была выделена ранее.
2. Перенаправить указатель arr на новый участок памяти (`arr = new_arr;`). Снова из-за использования ссылки, это перенаправление произойдёт и в main: внешний arr теперь будет указывать на новый выделенный массив.

Почему программа при этом не ругается:

• Мы сначала удаляем старый динамический массив (`delete[] arr`), освобождая занятую память.
• Затем перенаправляем указатель arr на новый массив (`arr = new_arr;`). Это безопасно, так как после удаления памяти arr уже не нужен для старого массива.
• Мы не используем arr до перенаправления (например, не обращаемся к старому массиву после удаления его), поэтому не получаем "висячий указатель" или ошибку доступа к памяти.

Таким образом, использование ссылки `int*& arr` в параметре позволяет нам управлять динамической памятью корректно и безопасно: мы можем удалять старую память и после этого перенаправлять указатель, а вся логика будет отражаться в "главном" указателе в main.

Когда вы создаёте указатель `new_arr` внутри функции, он "живёт" в этой функции, но память, на которую он указывает, выделена в куче (heap) через оператор `new[]`. После выполнения
arr = new_arr;

переменная `arr` (ссылка на указатель, фактически тот же указатель, что и в main) теперь указывает на эту только что выделенную память. 

После выхода из функции переменная `new_arr` как локальная больше не существует — но сама память остаётся "жива", потому что на неё теперь указывает переменная `arr` в основной программе (main). А значит, всё в порядке: у программы по-прежнему есть доступ к этим данным, и они не теряются. Указатель `new_arr` просто исчезает как переменная, а память не утечёт, так как есть актуальный arr. Потом память надо освободить там, где закончится работа с массивом (например, в конце main — через `delete[] arr;`).
