#include <string>

#include <iostream>

#include <algorithm> // Необходим для std::reverse [19, 20]

#include <memory>    // Для std::unique_ptr (рекомендация Best Practices)



// =========================================================

// ЗАДАНИЕ 1: ПАТТЕРН DECORATOR (Рендеринг HTML)

// =========================================================



/**

 * 1. Component (Интерфейс)

 * Определяет базовый контракт для всех текстовых объектов.

 * Виртуальный деструктор обязателен для полиморфного удаления.

 */

class Text {

public:

    virtual std::string render() const = 0;

    virtual ~Text() = default;

};



/**

 * 2. Concrete Component (Реализация)

 * Простой объект, хранящий строку. Ядро "луковицы" декораторов.

 */

class SimpleText : public Text {

    std::string text_;

public:

    explicit SimpleText(std::string t) : text_(std::move(t)) {}

    

    // override обеспечивает проверку сигнатуры компилятором [27]

    std::string render() const override {

        return text_;

    }

};



/**

 * 3. Base Decorator (Базовый класс декоратора)

 * Хранит ссылку на объект Text. Обратите внимание на использование raw pointer

 * для совместимости с условиями задачи, хотя std::unique_ptr был бы безопаснее.

 */

class Decorator : public Text {

protected:

    Text* ptr_; // Указатель на оборачиваемый объект

public:

    explicit Decorator(Text* ptr) : ptr_(ptr) {}

    

    std::string render() const override {

        return ptr_->render(); // Делегирование вызова

    }

    

    // Важно: В данной архитектуре владение объектом ptr_ не определено однозначно.

    // Клиентский код должен следить за удалением, либо нужно использовать smart pointers.

};



/**

 * Decorator 1: Paragraph

 * Оборачивает результат рендеринга в теги <p>.

 */

class Paragraph : public Decorator {

public:

    explicit Paragraph(Text* ptr) : Decorator(ptr) {}

    

    std::string render() const override {

        return "<p>" + ptr_->render() + "</p>";

    }

};



/**

 * Decorator 2: Reversed

 * Инвертирует строку. Это пример изменения данных, а не структуры.

 * Использует std::reverse для in-place инверсии копии строки.[20]

 */

class Reversed : public Decorator {

public:

    explicit Reversed(Text* ptr) : Decorator(ptr) {}

    

    std::string render() const override {

        std::string data = ptr_->render();

        std::reverse(data.begin(), data.end()); 

        return data;

    }

};



/**

 * Decorator 3: Link

 * Пример декоратора с дополнительным состоянием (url).

 * Сигнатура конструктора отличается от других декораторов, что допустимо.[28]

 */

class Link : public Decorator {

    std::string url_;

public:

    Link(Text* ptr, std::string url) : Decorator(ptr), url_(std::move(url)) {}

    

    std::string render() const override {

        return "<a href=" + url_ + ">" + ptr_->render() + "</a>";

    }

};



// =========================================================





// =========================================================

// ДЕМОНСТРАЦИЯ И ТЕСТИРОВАНИЕ

// =========================================================



int main() {

    // --- Тест 1: Decorator ---

    std::cout << "=== Decorator Pattern Test ===" << std::endl;

    

    // Создаем цепочку: Link -> Paragraph -> Reversed -> Text

    // Логика: Сначала текст реверсируется, затем оборачивается в параграф, затем в ссылку.

    // "Hello world" -> "dlrow olleH" -> "<p>dlrow olleH</p>" -> "<a href=...>...</a>"

    

    Text* text_obj = new SimpleText("Hello world");

    Text* reversed = new Reversed(text_obj);

    Text* paragraph = new Paragraph(reversed);

    Text* link = new Link(paragraph, "netology.ru");

    

    std::cout << link->render() << std::endl;

    

    // Очистка памяти (в обратном порядке создания оберток)

    // Примечание: При использовании std::unique_ptr это произошло бы автоматически.

    delete link; 

    delete paragraph; 

    delete reversed; 

    delete text_obj; 



   



    return 0;

}

Этот код — отличная иллюстрация классического паттерна Decorator. Давай разберем "под капот", чтобы понять, как именно работают наследование, конструкторы и сама цепочка вызовов.

Вот три ключевых механизма, которые заставляют это работать:

1. Наследование: "Быть" и "Иметь"
В паттерне Decorator есть хитрость: он использует наследование и композицию одновременно.

Наследование типа (Is-a): Все классы (SimpleText, Decorator, Paragraph, Reversed) наследуются от базового класса Text. Зачем? Чтобы каждый объект в системе гарантированно имел метод render(). Это позволяет нам передавать Paragraph в конструктор Link, потому что для компилятора они оба — просто Text*.

Агрегация (Has-a): Класс Decorator содержит поле Text* ptr_. Зачем? Чтобы хранить ссылку на следующий объект в цепочке.

2. Почему везде вызывается конструктор Decorator?
Ты заметил строчки вида : Decorator(ptr). Это список инициализации.

Давай посмотрим на цепочку создания объекта Paragraph:

C++
class Paragraph : public Decorator {
public:
    // Конструктор Paragraph принимает указатель на какой-то Text
    explicit Paragraph(Text* ptr) : Decorator(ptr) {} 
    //                              ^^^^^^^^^^^^^^
    //                              Вызов конструктора родителя!
Что здесь происходит:

Класс Paragraph сам не имеет поля для хранения указателя ptr.

Это поле (ptr_) находится в его родительском классе — Decorator.

Когда мы создаем Paragraph, мы обязаны передать полученный указатель «наверх» родителю, чтобы он сохранил его в ptr_.

Без этого вызова: Paragraph получил бы указатель, но не знал бы, куда его деть, а родительский Decorator остался бы с мусорным (неинициализированным) указателем.

3. Как работает цепочка (The Chain)
Представь это как луковицу или матрешку.

В main() ты создал такую структуру: Link -> (хранит внутри) Paragraph -> (хранит внутри) Reversed -> (хранит внутри) SimpleText ("Hello world").

Когда ты вызываешь link->render(), происходит "проваливание" внутрь и "всплытие" обратно.

Шаг 1: Погружение (Делегирование) Каждый объект говорит: "Я сделаю свою часть работы, но сначала дай мне результат от того, кто внутри меня".

Link: "Мне нужно обернуть всё в <a href...>, но сначала я вызываю ptr_->render() (это Paragraph)".

Paragraph: "Мне нужно обернуть всё в <p>, но сначала я вызываю ptr_->render() (это Reversed)".

Reversed: "Мне нужно перевернуть текст, но сначала я вызываю ptr_->render() (это SimpleText)".

SimpleText: "Внутри меня никого нет. Я просто возвращаю строку: 'Hello world'".

Шаг 2: Всплытие (Результат) Данные начинают возвращаться по цепочке, видоизменяясь на каждом шаге:

SimpleText вернул: "Hello world"

Reversed взял это, перевернул и вернул: "dlrow olleH"

Paragraph взял это, добавил теги и вернул: "<p>dlrow olleH</p>"

Link взял это, добавил ссылку и вернул финал: "<a href=netology.ru><p>dlrow olleH</p></a>"

Визуализация памяти
Когда ты делаешь new Link(paragraph, ...), в памяти это выглядит так:

Plaintext
[ Объект Link ]
   |
   +-- ptr_ ---------> [ Объект Paragraph ]
                          |
                          +-- ptr_ ---------> [ Объект Reversed ]
                                                 |
                                                 +-- ptr_ ---------> [ Объект SimpleText ]
                                                                        |
                                                                        +-- text_ = "Hello world"
Итог
Наследование нужно, чтобы все звенья цепи имели одинаковый интерфейс (Text).

Конструктор Decorator вызывается, чтобы сохранить указатель на следующее звено в единственном месте (в базовом классе), избегая дублирования кода.

Цепочка работает за счет того, что каждый метод render() вызывает render() у вложенного объекта, модифицирует результат и возвращает его наверх.

Почему при указание в конструкторе класса decorator он понимает что это например reversed

Это отличный вопрос, который затрагивает саму суть полиморфизма в C++.

Короткий ответ: Класс Decorator на самом деле НЕ знает, что это именно Reversed.

Ему это и не нужно знать. Ему достаточно знать, что переданный объект — это "нечто, что умеет делать render()".

Давай разберем это на примере механики языка.

1. Принцип "IS-A" (Является)
Смотри на иерархию наследования:

Reversed наследуется от Decorator.

Decorator наследуется от Text.

Вывод: Reversed — это Text.

Для компилятора это означает, что везде, где требуется Text* (указатель на Text), можно смело подсовывать Reversed*, Paragraph* или SimpleText*.

2. Магия указателя (Upcasting)
Посмотри на конструктор декоратора:

C++
// Конструктор ожидает ЛЮБОЙ Text
explicit Decorator(Text* ptr) : ptr_(ptr) {} 
А теперь на то, что происходит в main:

C++
// Здесь reversed - это указатель на тип Text (формально), 
// но указывает он на объект Reversed в памяти.
Text* reversed = new Reversed(text_obj); 

// Мы передаем 'reversed' в конструктор Paragraph.
// Конструктор Paragraph (через Decorator) сохраняет его в поле: Text* ptr_
Text* paragraph = new Paragraph(reversed); 
Внутри объекта Paragraph поле ptr_ имеет тип Text*. Но физически оно указывает на кусок памяти, где лежит объект Reversed.

3. Самое главное: Виртуальные функции (Virtual Dispatch)
Вот где происходит "магия", из-за которой вызывается правильный код.

В классе Text метод render() объявлен как virtual.

C++
virtual std::string render() const = 0;
Когда Paragraph вызывает ptr_->render(), происходит следующее:

Программа видит: "Ага, вызывается метод render() у указателя типа Text*".

Программа проверяет: "Этот метод виртуальный?" -> ДА.

Таблица виртуальных функций (vtable): Программа не смотрит на тип указателя (Text*). Она идет в память самого объекта, на который указывает стрелка.

В этом объекте (который по факту Reversed) есть скрытая таблица, где написано:

"Если у меня вызовут render, выполняй код из класса Reversed".

Аналогия из жизни
Представь, что ты — прораб (класс Decorator). Тебе нужен Рабочий (Text), чтобы забить гвоздь (render).

К тебе приходит человек. Ты не знаешь, кто он — плотник (Reversed), столяр (Paragraph) или стажер (SimpleText). Он надел каску, на которой написано просто "Рабочий". Тебе все равно. Ты просто даешь команду: "Забивай!".

Если под каской скрывается плотник, он забьет профессионально.

Если стажер — он забьет криво.

Тебе (Декоратору) не нужно знать имя и биографию рабочего. Тебе достаточно того, что он откликнулся на вызов "Рабочий" (наследование) и умеет выполнять команду "Забивай" (виртуальная функция).

Резюме
Декоратор "понимает" это не в момент компиляции, а в момент выполнения программы благодаря виртуальным функциям.

Он хранит ссылку как Text* (общий вид).

Когда вызывается метод, C++ смотрит на реальный тип объекта в памяти и запускает его версию метода.

Когда у конструктора link вызывается decorator(ptr) что происходит и как дальше идет цепочка

Давай разделим этот процесс на два этапа: сборка (конструктор) и работа (вызов render).

ЭТАП 1: Что происходит в конструкторе Link
Когда ты пишешь:

C++
new Link(paragraph, "netology.ru");
Вызывается конструктор класса Link. Вот его код:

C++
Link(Text* ptr, std::string url) : Decorator(ptr), url_(std::move(url)) {}
Вот пошаговая хронология событий в этой одной строке:

Запуск конструктора Link: Ты передал ему указатель ptr (который указывает на объект Paragraph).

Вызов Decorator(ptr): До того как Link начнет делать что-то свое, он обязан инициализировать свою родительскую часть (класс Decorator). Он передает этот указатель ptr наверх.

Работа конструктора Decorator:

Родительский класс берет этот указатель.

Кладет его в защищенное (protected) поле ptr_.

Итог: Теперь внутри объекта Link (в его родительской части) лежит ссылка на Paragraph.

Возврат в конструктор Link: Теперь Link инициализирует своё личное поле — url_ ("netology.ru").

Результат этапа сборки: У нас есть объект Link. Он "знает" (через поле ptr_), кто идет следующим (Paragraph), и знает свой URL.

ЭТАП 2: Как идет цепочка (Runtime)
Теперь самое интересное. Когда ты пишешь std::cout << link->render(), запускается цепная реакция.

Представь, что это эстафета. Каждый бегун делает свою часть работы, но только после того, как получит палочку от предыдущего. В коде это выглядит как стек вызовов.

Шаг 1: Link (Начальник)
Вызывается Link::render().

Код: return "<a href=" + url_ + ">" + ptr_->render() + "</a>";

Действие: Link видит, что ему нужна "начинка" (середина тега).

Событие: Он дергает ptr_->render(). Так как в ptr_ лежит Paragraph, управление передается туда.

Link замирает и ждет результата...

Шаг 2: Paragraph (Среднее звено)
Вызывается Paragraph::render().

Код: return "<p>" + ptr_->render() + "</p>";

Действие: Paragraph тоже нужна "начинка".

Событие: Он дергает ptr_->render(). В его ptr_ (который мы настроили ранее) лежит Reversed.

Paragraph замирает и ждет...

Шаг 3: Reversed (Трюкач)
Вызывается Reversed::render().

Код:

C++
std::string data = ptr_->render(); // Сначала дай мне текст!
std::reverse(data.begin(), data.end()); // Потом я его переверну
return data;
Событие: Он дергает ptr_->render(). В его ptr_ лежит SimpleText.

Reversed замирает и ждет...

Шаг 4: SimpleText (Ядро)
Вызывается SimpleText::render().

Код: return text_;

Действие: У него нет ptr_. Ему некого ждать. Он просто берет строку "Hello world".

Событие: ВОЗВРАТ (return) "Hello world".

Обратный путь (Сборка результата)
Теперь цепочка "схлопывается" обратно, снизу вверх:

SimpleText вернул "Hello world".

Reversed проснулся. Получил "Hello world".

Перевернул: "dlrow olleH".

Вернул это в Paragraph.

Paragraph проснулся. Получил "dlrow olleH".

Приклеил теги: "<p>dlrow olleH</p>".

Вернул это в Link.

Link проснулся. Получил "<p>dlrow olleH</p>".

Приклеил тег ссылки: "<a href=netology.ru><p>dlrow olleH</p></a>".

Вернул это в main.

В консоль выводится итоговая строка.

Главный секрет
Цепочка работает только потому, что каждый декоратор вызывает ptr_->render(). Если бы кто-то (например, Link) забыл это сделать и просто вернул строку, цепочка бы оборвалась, и всё, что было внутри (Paragraph, Reversed, Text), было бы проигнорировано.

 auto text_block = new ItalicText(new BoldText(new Text())); // последовательно запустить

Короткий ответ: Она работает логически (текст выведется), но создает серьезную утечку памяти.

Проблема не в том, как строится цепочка, а в том, как мы теряем доступ к внутренним объектам, которые создаем через new.

Давай разберем, почему вариант auto text_block = new ... в данном конкретном коде — это ошибка.

1. Что происходит в памяти
Когда ты пишешь вложенную конструкцию:

C++
// (1) Создается Text -> (2) Создается BoldText -> (3) Создается ItalicText
auto text_block = new ItalicText(new BoldText(new Text()));
У тебя в переменной text_block сохраняется адрес только самой внешней оболочки (ItalicText).

Адрес BoldText? Потерян. Он лежит только внутри ptr_ у ItalicText.

Адрес Text? Потерян. Он лежит только внутри ptr_ у BoldText.

2. Почему это проблема (Утечка памяти)
В твоем коде класс Decorator выглядит так:

C++
class Decorator : public Text {
protected:
    Text* ptr_; 
public:
    // ... конструктор ...
    // Деструктора НЕТ (используется default), либо он пустой!
};
Это значит, что когда ты захочешь удалить объект:

C++
delete text_block; // Удаляет только ItalicText!
Вызывается деструктор ItalicText (он пустой).

Память, занятая под ItalicText, освобождается.

НО: Деструктор не вызывает delete ptr_.

Результат: BoldText и Text остаются висеть в оперативной памяти навсегда (до закрытия программы). У тебя нет указателей на них, чтобы удалить их вручную.

Как это исправить?
Есть два пути.

Путь А: "Олдскульный" (Меняем класс Decorator)
Нужно научить Декоратор владеть объектом. Мы добавляем деструктор, который удаляет вложенный объект.

C++
class Decorator : public Text {
    // ... поля ...
public:
    virtual ~Decorator() {
        delete ptr_; // Теперь удаление пойдет по цепочке вниз!
    }
};
Теперь delete text_block удалит ptr_, тот удалит свой ptr_, и всё очистится.

Путь Б: "Современный" (Smart Pointers) — Рекомендуемый
В современном C++ (начиная с C++11) мы не используем new и сырые указатели вот так. Мы используем std::unique_ptr. Это позволяет писать вложенные конструкции безопасно.

C++
#include <memory>

// Меняем типы в конструкторах на std::unique_ptr<Text>
auto text_block = std::make_unique<ItalicText>(
    std::make_unique<BoldText>(
        std::make_unique<Text>()
    )
);

// render сработает
std::cout << text_block->render();

// delete НЕ НУЖЕН. Память очистится сама, когда text_block выйдет из области видимости.
Резюме
Твоя строка "не работает" не потому, что C++ не понимает вложенность, а потому что в твоей текущей реализации классов нет механизма автоматической очистки вложенных объектов.

В первом примере (main из задания) ты сохранял каждый указатель в отдельную переменную (reversed, paragraph...), поэтому мог удалить их вручную по очереди. Во вложенной записи у тебя нет доступа к "внутренностям", и они превращаются в мусор.
