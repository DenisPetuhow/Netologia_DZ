#include <iostream>
#include <Wt/Dbo/Dbo.h>
#include <Wt/Dbo/backend/Postgres.h>
#include <Windows.h> //для SetConsoleOutputCP

#pragma execution_character_set ("utf-8") //строковые литералы в UTF-8

class Publisher;

using namespace std;
using namespace Wt::Dbo;

//Просто книга с названием
class Book {
public:
    std::string	title;
    //ptr<Publisher>	publisher; //TODO: добавить самостоятельно связь
    //collection < ptr<Stock> > stocks; //TODO: добавить самостоятельно связь 

    template <typename Action>
    void persist(Action& a) {
        field(a, title, "title");
    }
};

int main(int argc, char** argv) {
    SetConsoleOutputCP(CP_UTF8); //вывод на консоль в UTF-8

    try {
        //коннектимся к базе
        string connectionString = //введите ваши данные для подключения
            "host=localhost "
            "port=5432 "
            "user=postgres "
            "password= "
            "client_encoding=UTF8 ";

        unique_ptr<backend::Postgres> connection{ new backend::Postgres(connectionString) };
        Session session;
        session.setConnection(move(connection));

        session.mapClass<Book>("Book");

        //Создание таблиц, если их ещё не было
        session.createTables();
        cout << "Create tables ok!\n";
        
	    //выполнение запроса...
    }
    catch (const std::exception& e) {
        cout << e.what() << endl;
    }

    return 0;
}
